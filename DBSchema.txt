-- USERS
CREATE TABLE IF NOT EXISTS public.users (
    id UUID NOT NULL DEFAULT gen_random_uuid(),   
    email VARCHAR(50) NOT NULL,
    hashedpassword VARCHAR(255) NOT NULL,
    CONSTRAINT users_pkey PRIMARY KEY (id),
    CONSTRAINT users_email_key UNIQUE (email)
);

-- PRODUCTS
CREATE TABLE IF NOT EXISTS public.products (
    id UUID NOT NULL DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,
    category VARCHAR(20) NOT NULL,
    stock NUMERIC(10,3) NOT NULL,
    unit VARCHAR(20) NOT NULL,
    price NUMERIC(10,2) NOT NULL,
    min_stock INTEGER NOT NULL,
    last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT products_pkey PRIMARY KEY (id)
);

-- EMPLOYEES
CREATE TABLE IF NOT EXISTS public.employees (
    id UUID NOT NULL DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,
    phone VARCHAR(50) NOT NULL,
    position VARCHAR(50) NOT NULL,
    salary NUMERIC(10,2) NOT NULL,
    advances NUMERIC(10,2) DEFAULT 0,
    bonus NUMERIC(10,2) DEFAULT 0,
    penalties NUMERIC(10,2) DEFAULT 0,
    hire_date DATE NOT NULL,
    net_salary NUMERIC(10,2) DEFAULT 0,
    CONSTRAINT employees_pkey PRIMARY KEY (id)
);

-- SUPPLIERS
CREATE TABLE IF NOT EXISTS public.suppliers (
    id UUID NOT NULL DEFAULT gen_random_uuid(),
    name VARCHAR(50) NOT NULL,
    phone VARCHAR(50) NOT NULL,
    email VARCHAR(50) NOT NULL,
    address VARCHAR(100) NOT NULL,
    total_debt NUMERIC(15,2) DEFAULT 0,
    total_paid NUMERIC(15,2) DEFAULT 0,
    CONSTRAINT suppliers_pkey PRIMARY KEY (id),
    CONSTRAINT suppliers_email_key UNIQUE (email)
);

-- TRANSACTION TYPE ENUM
CREATE TYPE transaction_type AS ENUM ('purchase', 'sale', 'expense');

-- TRANSACTIONS 
CREATE TABLE IF NOT EXISTS public.transactions (
    id UUID NOT NULL DEFAULT gen_random_uuid(),
    type transaction_type NOT NULL,
    amount integer NOT NULL,
    description VARCHAR(5000),
    date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
);
